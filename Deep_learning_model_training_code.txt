# -*- coding: utf-8 -*-
"""
Bounded log-space neural network for predicting carrier kinetic parameters
from combined TrPL and PLQY data.

This script performs the following tasks:
1. Reads TrPL + PLQY data from an Excel file.
2. Uses the first six columns as target kinetic parameters.
3. Uses the remaining columns as input features.
4. Applies StandardScaler to input features.
5. Converts target parameters to log10 space.
6. Trains a bounded MLP model whose outputs are constrained within
   physically defined parameter ranges.
7. Evaluates model performance on training and test sets.
8. Saves prediction results, model weights, scaler, and boundary information.
9. Generates a 2 × 3 hexbin correlation plot for predicted vs. labeled values.

Author: [Wenning Chen et al.]
"""

# ==========================================================
# 1. Import packages
# ==========================================================
import random
import joblib

import numpy as np
import pandas as pd
import matplotlib
import matplotlib.pyplot as plt

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader, random_split

from sklearn.preprocessing import StandardScaler
from sklearn.metrics import r2_score
from matplotlib.ticker import ScalarFormatter


# ==========================================================
# 2. Global settings
# ==========================================================
SEED = 42

INPUT_FILE = "TrPL+PLQY.xlsx"

TRAIN_RESULT_FILE = "train_prediction_results_bounded_log.csv"
TEST_RESULT_FILE = "test_prediction_results_bounded_log.csv"

MODEL_FILE = "trained_bounded_log_model.pth"
X_SCALER_FILE = "x_scaler.pkl"
BOUNDS_INFO_FILE = "bounds_info.pkl"

FIGURE_FILE = "correlation_hexbin_plot_2x3_train_test_R2_checked.png"

BATCH_SIZE = 32
EPOCHS = 800
LEARNING_RATE = 0.000005
DROPOUT_RATE = 0.048
HIDDEN_DIM = 1800
TRAIN_RATIO = 0.8

matplotlib.rcParams["font.family"] = "Arial"


# ==========================================================
# 3. Reproducibility
# ==========================================================
def set_random_seed(seed: int = 42):
    """
    Set random seed for NumPy, Python random, and PyTorch.

    Parameters
    ----------
    seed : int
        Random seed.
    """
    random.seed(seed)
    np.random.seed(seed)

    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)


# ==========================================================
# 4. Dataset definition
# ==========================================================
class EquationDataset(Dataset):
    """
    PyTorch dataset for input features and log-space target parameters.
    """

    def __init__(self, x_tensor: torch.Tensor, y_tensor: torch.Tensor):
        self.x_tensor = x_tensor
        self.y_tensor = y_tensor

    def __len__(self):
        return len(self.x_tensor)

    def __getitem__(self, index):
        return self.x_tensor[index], self.y_tensor[index]


# ==========================================================
# 5. Bounded MLP model
# ==========================================================
class BoundedMLP(nn.Module):
    """
    Multi-layer perceptron with bounded log-space outputs.

    The model predicts six log10-transformed kinetic parameters:
        log10(Rpop), log10(Rdep), log10(NT),
        log10(Reh), log10(Rdet), log10(RAug)

    A sigmoid activation is applied to the raw output and then mapped to
    the physically allowed log10 parameter range.
    """

    def __init__(
        self,
        input_dim: int,
        lower_log: np.ndarray,
        upper_log: np.ndarray,
        hidden_dim: int = 1800,
        dropout_rate: float = 0.048
    ):
        super(BoundedMLP, self).__init__()

        self.feature_net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout_rate),

            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout_rate),

            nn.Linear(hidden_dim, 6)
        )

        # Register the parameter bounds as buffers so that they move
        # automatically with the model between CPU and GPU.
        self.register_buffer(
            "lower_log",
            torch.tensor(lower_log, dtype=torch.float32)
        )

        self.register_buffer(
            "upper_log",
            torch.tensor(upper_log, dtype=torch.float32)
        )

    def forward(self, x):
        raw_output = self.feature_net(x)

        # Sigmoid constrains the normalized output to [0, 1].
        normalized_output = torch.sigmoid(raw_output)

        # Map normalized output to the predefined log10 parameter range.
        bounded_log_output = (
            self.lower_log
            + normalized_output * (self.upper_log - self.lower_log)
        )

        return bounded_log_output


# ==========================================================
# 6. Data loading and preprocessing
# ==========================================================
def load_and_preprocess_data(input_file: str):
    """
    Load data, split input/output columns, scale X, and log-transform Y.

    Parameters
    ----------
    input_file : str
        Path to the Excel file.

    Returns
    -------
    X : torch.Tensor
        Scaled input features.
    Y : torch.Tensor
        log10-transformed target parameters.
    x_scaler : StandardScaler
        Fitted scaler for input features.
    output_columns : list
        Names of the six output parameters.
    Y_raw : np.ndarray
        Original target parameters.
    X_raw : np.ndarray
        Original input features.
    """

    data = pd.read_excel(input_file)

    # First six columns are target kinetic parameters.
    Y_raw = data.iloc[:, :6].values.astype(np.float64)

    # Remaining columns are input features, such as TrPL and PLQY data.
    X_raw = data.iloc[:, 6:].values.astype(np.float64)

    output_columns = data.columns[:6].tolist()

    print("Output columns:", output_columns)
    print("X shape:", X_raw.shape)
    print("Y shape:", Y_raw.shape)

    # Check whether all target values are positive before log10 transform.
    if np.any(Y_raw <= 0):
        raise ValueError(
            "Y contains values <= 0, so log10 transformation cannot be applied. "
            "Please check the first six columns of the Excel file."
        )

    x_scaler = StandardScaler()
    X_scaled = x_scaler.fit_transform(X_raw)

    Y_log = np.log10(Y_raw)

    X = torch.tensor(X_scaled, dtype=torch.float32)
    Y = torch.tensor(Y_log, dtype=torch.float32)

    return X, Y, x_scaler, output_columns, Y_raw, X_raw


# ==========================================================
# 7. Parameter bounds
# ==========================================================
def define_parameter_bounds():
    """
    Define the physical lower and upper bounds of the six kinetic parameters.

    The order must be consistent with the first six columns in the Excel file.

    Returns
    -------
    param_lower : np.ndarray
        Lower bounds in original physical scale.
    param_upper : np.ndarray
        Upper bounds in original physical scale.
    lower_log : np.ndarray
        Lower bounds in log10 scale.
    upper_log : np.ndarray
        Upper bounds in log10 scale.
    """

    param_lower = np.array([
        1e-10,   # Rec
        1e-11,   # Rhc
        1e14,    # NT
        1e-12,   # Reh
        1e4,     # Ree
        1e-29    # RAug
    ], dtype=np.float64)

    param_upper = np.array([
        1e-8,    # Rec
        1e-9,    # Rhc
        1e16,    # NT
        1e-10,   # Reh
        1e6,     # Ree
        1e-27    # RAug
    ], dtype=np.float64)

    lower_log = np.log10(param_lower)
    upper_log = np.log10(param_upper)

    print("\nlog10 lower bound:", lower_log)
    print("log10 upper bound:", upper_log)

    return param_lower, param_upper, lower_log, upper_log


def check_parameter_ranges(
    Y_raw: np.ndarray,
    output_columns: list,
    param_lower: np.ndarray,
    param_upper: np.ndarray
):
    """
    Check whether target values fall within predefined physical ranges.

    Parameters
    ----------
    Y_raw : np.ndarray
        Original target parameters.
    output_columns : list
        Names of output parameters.
    param_lower : np.ndarray
        Lower bounds in original scale.
    param_upper : np.ndarray
        Upper bounds in original scale.
    """

    print("\nChecking target parameter ranges:")

    for i, col in enumerate(output_columns):
        y_min = Y_raw[:, i].min()
        y_max = Y_raw[:, i].max()

        print(f"\n{col}:")
        print(f"  data min = {y_min:.6e}")
        print(f"  data max = {y_max:.6e}")
        print(f"  allowed  = [{param_lower[i]:.6e}, {param_upper[i]:.6e}]")

        if y_min < param_lower[i] or y_max > param_upper[i]:
            print("  WARNING: Some values are outside the predefined range.")


# ==========================================================
# 8. Train/test split and dataloaders
# ==========================================================
def create_dataloaders(
    X: torch.Tensor,
    Y: torch.Tensor,
    train_ratio: float = 0.8,
    batch_size: int = 32,
    seed: int = 42
):
    """
    Create training and test dataloaders.

    Parameters
    ----------
    X : torch.Tensor
        Input feature tensor.
    Y : torch.Tensor
        Target tensor in log10 space.
    train_ratio : float
        Fraction of data used for training.
    batch_size : int
        Batch size.
    seed : int
        Random seed for reproducible splitting.

    Returns
    -------
    train_loader : DataLoader
        Training dataloader with shuffle.
    train_eval_loader : DataLoader
        Training dataloader without shuffle for evaluation.
    test_loader : DataLoader
        Test dataloader.
    """

    dataset = EquationDataset(X, Y)

    train_size = int(train_ratio * len(dataset))
    test_size = len(dataset) - train_size

    train_dataset, test_dataset = random_split(
        dataset,
        [train_size, test_size],
        generator=torch.Generator().manual_seed(seed)
    )

    train_loader = DataLoader(
        train_dataset,
        batch_size=batch_size,
        shuffle=True
    )

    train_eval_loader = DataLoader(
        train_dataset,
        batch_size=batch_size,
        shuffle=False
    )

    test_loader = DataLoader(
        test_dataset,
        batch_size=batch_size,
        shuffle=False
    )

    return train_loader, train_eval_loader, test_loader


# ==========================================================
# 9. Model training
# ==========================================================
def train_model(
    model: nn.Module,
    train_loader: DataLoader,
    device: torch.device,
    epochs: int = 800,
    learning_rate: float = 0.000005
):
    """
    Train the bounded MLP model in log10 parameter space.

    Parameters
    ----------
    model : nn.Module
        Bounded MLP model.
    train_loader : DataLoader
        Training dataloader.
    device : torch.device
        CPU or CUDA device.
    epochs : int
        Number of training epochs.
    learning_rate : float
        Learning rate.
    """

    model.to(device)

    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)

    for epoch in range(epochs):
        model.train()
        total_loss = 0.0

        for X_batch, Y_batch in train_loader:
            X_batch = X_batch.to(device)
            Y_batch = Y_batch.to(device)

            optimizer.zero_grad()

            outputs_log = model(X_batch)
            loss = criterion(outputs_log, Y_batch)

            loss.backward()
            optimizer.step()

            total_loss += loss.item()

        avg_loss = total_loss / len(train_loader)

        print(f"Epoch [{epoch + 1}/{epochs}], Loss: {avg_loss:.8f}")


# ==========================================================
# 10. Prediction, inverse transform, R2 calculation, and saving
# ==========================================================
def predict_and_save(
    model: nn.Module,
    data_loader: DataLoader,
    device: torch.device,
    save_filename: str,
    dataset_name: str,
    output_columns: list,
    lower_log: np.ndarray,
    upper_log: np.ndarray
):
    """
    Predict kinetic parameters, calculate R2, and save prediction results.

    Parameters
    ----------
    model : nn.Module
        Trained model.
    data_loader : DataLoader
        DataLoader for training or test set.
    device : torch.device
        CPU or CUDA device.
    save_filename : str
        Output CSV filename.
    dataset_name : str
        Name of the dataset for printing.
    output_columns : list
        Names of output parameters.
    lower_log : np.ndarray
        Lower bounds in log10 scale.
    upper_log : np.ndarray
        Upper bounds in log10 scale.

    Returns
    -------
    all_labels_original : np.ndarray
        True target values in original scale.
    all_preds_original : np.ndarray
        Predicted target values in original scale.
    all_labels_log : np.ndarray
        True target values in log10 scale.
    all_preds_log : np.ndarray
        Predicted target values in log10 scale.
    r2_scores_original : list
        R2 values in original scale.
    r2_scores_log : list
        R2 values in log10 scale.
    """

    model.eval()
    model.to(device)

    criterion = nn.MSELoss()
    total_loss = 0.0

    all_preds_log = []
    all_labels_log = []

    with torch.no_grad():
        for X_batch, Y_batch in data_loader:
            X_batch = X_batch.to(device)
            Y_batch = Y_batch.to(device)

            outputs_log = model(X_batch)

            loss = criterion(outputs_log, Y_batch)
            total_loss += loss.item()

            all_preds_log.append(outputs_log.cpu())
            all_labels_log.append(Y_batch.cpu())

    avg_loss = total_loss / len(data_loader)
    print(f"\n{dataset_name} Log-space Loss: {avg_loss:.8f}")

    all_preds_log = torch.cat(all_preds_log, dim=0).numpy().astype(np.float64)
    all_labels_log = torch.cat(all_labels_log, dim=0).numpy().astype(np.float64)

    # The model output is already bounded. This clipping is only used as
    # an additional numerical safeguard.
    all_preds_log = np.clip(all_preds_log, lower_log, upper_log)

    # Convert log10 values back to original physical scale.
    all_preds_original = 10 ** all_preds_log
    all_labels_original = 10 ** all_labels_log

    r2_scores_original = []
    r2_scores_log = []

    print(f"\n{dataset_name} R² Scores:")

    for i in range(6):
        y_true_original = all_labels_original[:, i].astype(np.float64)
        y_pred_original = all_preds_original[:, i].astype(np.float64)

        y_true_log = all_labels_log[:, i].astype(np.float64)
        y_pred_log = all_preds_log[:, i].astype(np.float64)

        r2_original = r2_score(y_true_original, y_pred_original)
        r2_log = r2_score(y_true_log, y_pred_log)

        r2_scores_original.append(r2_original)
        r2_scores_log.append(r2_log)

        print(f"{output_columns[i]}:")
        print(f"  R2 original scale = {r2_original:.10f}")
        print(f"  R2 log10 scale    = {r2_log:.10f}")

    # Save true and predicted values.
    result_dict = {}

    for i, col_name in enumerate(output_columns):
        result_dict[f"True_{col_name}"] = all_labels_original[:, i]
        result_dict[f"Pred_{col_name}"] = all_preds_original[:, i]

        result_dict[f"True_log10_{col_name}"] = all_labels_log[:, i]
        result_dict[f"Pred_log10_{col_name}"] = all_preds_log[:, i]

    result_df = pd.DataFrame(result_dict)
    result_df.to_csv(save_filename, index=False, encoding="utf-8-sig")

    print(f"\n{dataset_name} prediction results saved to: {save_filename}")

    return (
        all_labels_original,
        all_preds_original,
        all_labels_log,
        all_preds_log,
        r2_scores_original,
        r2_scores_log
    )


# ==========================================================
# 11. Plot 2 × 3 correlation hexbin figure
# ==========================================================
def plot_correlation_hexbin(
    train_labels: np.ndarray,
    train_preds: np.ndarray,
    test_labels: np.ndarray,
    test_preds: np.ndarray,
    output_columns: list,
    save_filename: str
):
    """
    Generate a 2 × 3 hexbin correlation plot for predicted vs. labeled values.

    Parameters
    ----------
    train_labels : np.ndarray
        Training set true values in original scale.
    train_preds : np.ndarray
        Training set predicted values in original scale.
    test_labels : np.ndarray
        Test set true values in original scale.
    test_preds : np.ndarray
        Test set predicted values in original scale.
    output_columns : list
        Names of output parameters.
    save_filename : str
        Output image filename.
    """

    fig, axes = plt.subplots(
        2,
        3,
        figsize=(15, 8),
        constrained_layout=True
    )

    axes = axes.flatten()

    hb = None

    for i, ax in enumerate(axes[:6]):

        ax.set_axisbelow(True)
        ax.grid(True, linestyle="--", alpha=1, zorder=0)

        # Current target parameter.
        y_train_true = train_labels[:, i].astype(np.float64)
        y_train_pred = train_preds[:, i].astype(np.float64)

        y_test_true = test_labels[:, i].astype(np.float64)
        y_test_pred = test_preds[:, i].astype(np.float64)

        # Recalculate R2 values for plotting.
        train_r2_now = r2_score(y_train_true, y_train_pred)
        test_r2_now = r2_score(y_test_true, y_test_pred)

        # Test set hexbin plot.
        hb = ax.hexbin(
            y_test_true,
            y_test_pred,
            gridsize=1000,
            cmap="RdBu_r",
            mincnt=1,
            vmin=1,
            vmax=5
        )

        # Determine plot limits using both training and test values.
        all_values = np.concatenate([
            y_train_true,
            y_train_pred,
            y_test_true,
            y_test_pred
        ])

        vmin = np.nanmin(all_values)
        vmax = np.nanmax(all_values)

        # y = x reference line.
        ax.plot(
            [vmin, vmax],
            [vmin, vmax],
            color="black",
            linestyle="-",
            linewidth=1.5,
            zorder=3
        )

        ax.set_xlim(vmin, vmax)
        ax.set_ylim(vmin, vmax)

        # Axis labels.
        ax.set_xlabel("Labeled", fontsize=20)
        ax.set_ylabel("Predicted", fontsize=20)

        # R2 annotation.
        ax.text(
            0.05,
            0.95,
            f"{output_columns[i]}\n"
            f"Train $R^2$ = {train_r2_now:.4f}\n"
            f"Test $R^2$ = {test_r2_now:.4f}",
            transform=ax.transAxes,
            fontsize=15,
            verticalalignment="top",
            horizontalalignment="left",
            bbox=dict(
                facecolor="white",
                edgecolor="none",
                alpha=0.75,
                boxstyle="round,pad=0.25"
            )
        )

        # Axis formatting.
        ax.tick_params(
            axis="both",
            labelsize=16,
            direction="in"
        )

        ax.locator_params(axis="x", nbins=5)
        ax.locator_params(axis="y", nbins=5)

        ax.xaxis.set_major_formatter(ScalarFormatter())
        ax.yaxis.set_major_formatter(ScalarFormatter())

        ax.ticklabel_format(
            style="sci",
            scilimits=(-2, 2),
            axis="both"
        )

        ax.xaxis.offsetText.set_fontsize(14)
        ax.yaxis.offsetText.set_fontsize(14)

    # Shared colorbar.
    cbar = fig.colorbar(hb, ax=axes, shrink=0.9)
    cbar.set_label("Density", fontsize=18)
    cbar.ax.tick_params(labelsize=14)

    plt.savefig(
        save_filename,
        dpi=600,
        bbox_inches="tight"
    )

    plt.show()

    print(f"\nCorrelation hexbin figure saved to: {save_filename}")


# ==========================================================
# 12. Save model, scaler, and boundary information
# ==========================================================
def save_model_and_metadata(
    model: nn.Module,
    x_scaler: StandardScaler,
    output_columns: list,
    param_lower: np.ndarray,
    param_upper: np.ndarray,
    lower_log: np.ndarray,
    upper_log: np.ndarray
):
    """
    Save trained model, input scaler, and parameter boundary information.
    """

    torch.save(model.state_dict(), MODEL_FILE)
    joblib.dump(x_scaler, X_SCALER_FILE)

    bounds_info = {
        "output_columns": output_columns,
        "param_lower": param_lower,
        "param_upper": param_upper,
        "lower_log": lower_log,
        "upper_log": upper_log
    }

    joblib.dump(bounds_info, BOUNDS_INFO_FILE)

    print(f"\nModel saved to: {MODEL_FILE}")
    print(f"X scaler saved to: {X_SCALER_FILE}")
    print(f"Bounds info saved to: {BOUNDS_INFO_FILE}")


# ==========================================================
# 13. Main workflow
# ==========================================================
def main():
    """
    Main workflow for training, evaluation, saving, and plotting.
    """

    set_random_seed(SEED)

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print("\nUsing device:", device)

    # Define physical parameter bounds.
    param_lower, param_upper, lower_log, upper_log = define_parameter_bounds()

    # Load and preprocess data.
    X, Y, x_scaler, output_columns, Y_raw, X_raw = load_and_preprocess_data(
        INPUT_FILE
    )

    # Check whether the target parameters are inside the predefined ranges.
    check_parameter_ranges(
        Y_raw=Y_raw,
        output_columns=output_columns,
        param_lower=param_lower,
        param_upper=param_upper
    )

    # Create dataloaders.
    train_loader, train_eval_loader, test_loader = create_dataloaders(
        X=X,
        Y=Y,
        train_ratio=TRAIN_RATIO,
        batch_size=BATCH_SIZE,
        seed=SEED
    )

    # Initialize bounded MLP model.
    model = BoundedMLP(
        input_dim=X.shape[1],
        lower_log=lower_log,
        upper_log=upper_log,
        hidden_dim=HIDDEN_DIM,
        dropout_rate=DROPOUT_RATE
    )

    # Train model.
    train_model(
        model=model,
        train_loader=train_loader,
        device=device,
        epochs=EPOCHS,
        learning_rate=LEARNING_RATE
    )

    # Evaluate and save training set predictions.
    (
        train_labels,
        train_preds,
        train_labels_log,
        train_preds_log,
        train_r2_original,
        train_r2_log
    ) = predict_and_save(
        model=model,
        data_loader=train_eval_loader,
        device=device,
        save_filename=TRAIN_RESULT_FILE,
        dataset_name="Train Set",
        output_columns=output_columns,
        lower_log=lower_log,
        upper_log=upper_log
    )

    # Evaluate and save test set predictions.
    (
        test_labels,
        test_preds,
        test_labels_log,
        test_preds_log,
        test_r2_original,
        test_r2_log
    ) = predict_and_save(
        model=model,
        data_loader=test_loader,
        device=device,
        save_filename=TEST_RESULT_FILE,
        dataset_name="Test Set",
        output_columns=output_columns,
        lower_log=lower_log,
        upper_log=upper_log
    )

    # Save trained model and preprocessing information.
    save_model_and_metadata(
        model=model,
        x_scaler=x_scaler,
        output_columns=output_columns,
        param_lower=param_lower,
        param_upper=param_upper,
        lower_log=lower_log,
        upper_log=upper_log
    )

    # Plot correlation hexbin figure.
    plot_correlation_hexbin(
        train_labels=train_labels,
        train_preds=train_preds,
        test_labels=test_labels,
        test_preds=test_preds,
        output_columns=output_columns,
        save_filename=FIGURE_FILE
    )


# ==========================================================
# 14. Run script
# ==========================================================
if __name__ == "__main__":
    main()
